### Info
:::info

Name: **NanoByte.CloneGenerator**

Roslyn source generator for deep Clone() methods. Annotate a partial class with [Cloneable] and the member-by-member copy is generated at compile time.

// <auto-generated/>
#nullable enable
#pragma warning disable CS0618 // Member is obsolete

namespace CloneData
{
    partial class Person
    {
        /// <summary>Copies all members declared by <see cref="Person"/> from one instance to another.</summary>
        protected static void CloneFromTo(global::CloneData.Person from, global::CloneData.Person to)
        {
            if (from == null) throw new global::System.ArgumentNullException(nameof(from));
            if (to == null) throw new global::System.ArgumentNullException(nameof(to));
            to.FirstName = from.FirstName;
            to.LastName = from.LastName;
            to.Age = from.Age;
            to.Childs = from.Childs;
        }

        /// <summary>Creates a deep copy of this <see cref="Person"/> instance.</summary>
        /// <returns>The new copy of the <see cref="Person"/>.</returns>
        public virtual global::CloneData.Person Clone()
        {
            var from = this;
            var to = new global::CloneData.Person();
            CloneFromTo(from, to);
            return to;
        }
    }
}
